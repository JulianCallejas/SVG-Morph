Usage Examples

// Basic usage
const morphAnimation = createSVGMorph('myContainer');
morphAnimation.play();

// With custom options
const morphAnimation = createSVGMorph('myContainer', {
  duration: 3000,
  easing: 'bounce',
  loop: true,
  onComplete: () => console.log('Animation complete!'),
  onUpdate: (progress) => console.log('Progress:', progress)
});

// Manual control
morphAnimation.setProgress(0.5); // Set to 50%
morphAnimation.reverse();        // Reverse direction
morphAnimation.restart();        // Restart from beginning
